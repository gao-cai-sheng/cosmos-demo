      import { createMarsWeather } from '../src/weather/mars-weather.js';

const wx = createMarsWeather({ seed: 1, elevationKm });
const time = { sol, dayFraction, refLon };   // 调用方自己的时钟

card.update(wx.sample(lat, lon, time));       // HUD 天气卡
wx.route(waypoints, time).crossings;          // 航线穿越哪些尘暴
drawStormLayer(ctx, wx.storms(time), { toScreen, pxPerDeg, width, height });
